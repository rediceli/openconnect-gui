#!/usr/bin/env python3
"""
Minimal AnyConnect-compatible mock gateway for P0 validation.

Implements just enough of the Cisco AnyConnect XML flow that openconnect
gets past authentication:

  POST type="init"      -> <config-auth type="auth-reply"> containing the
                           login form (username / password / auth_group select)
  POST type="auth-reply"-> same doc back with the submitted values, or an
                           error <message>, or on success a session cookie +
                           <auth id="100" method="none"/> + <config>

It does NOT implement DTLS/CSTP, so no real tunnel. That is intentional:
we only need openconnect's log lines, which is what a state parser consumes.
"""
import http.server
import os
import re
import ssl
import sys

PASSWORD = os.environ.get("MOCK_PASSWORD", "hunter2")
PORT = int(os.environ.get("MOCK_PORT", "8443"))

HDR = '<?xml version="1.0" encoding="UTF-8" ?>\n'

FORM = HDR + """<config-auth client="vpn" type="auth-reply">
 <version who="vpn">v9.21</version>
 <device-id type="desktop" platform="MacIntel">mockhost.local</device-id>
 <auth id="main">
  <title>Login</title>
  <message>Please enter your username and password.</message>
  <form>
   <input type="text" name="username" label="Username:"/>
   <input type="password" name="password" label="Password:"/>
   <select name="group_list" label="GROUP:">
    <option selected="none"></option>
    <option>Engineering</option>
    <option>Operations</option>
   </select>
  </form>
 </auth>
</config-auth>
"""

AUTH_FAIL = HDR + """<config-auth client="vpn" type="auth-reply">
 <auth id="main">
  <title>Login</title>
  <message id="0">Invalid username or password</message>
  <message id="1">Please try again</message>
 </auth>
</config-auth>
"""

AUTH_OK = HDR + """<config-auth client="vpn" type="auth-reply">
 <auth id="success">
  <banner></banner>
  <message id="0">Please wait while the VPN connection is being established.</message>
  <session-timeout>14400</session-timeout>
 </auth>
</config-auth>
"""

CONFIG_REPLY = HDR + """<config client="vpn" type="private">
 <version who="vpn">v9.21</version>
 <device-id type="desktop" platform="MacIntel">mockhost.local</device-id>
 <vpn-base-config>
  <server-cert-hash>SHA256:MOCKCERTFINGERPRINT</server-cert-hash>
 </vpn-base-config>
 <vpn-base-config>
  <local-address>10.99.99.2</local-address>
  <remote-address>10.99.99.1</remote-address>
  <dns-server>10.99.99.53</dns-server>
  <mtu>1400</mtu>
 </vpn-base-config>
</config>
"""


def tag(text, name):
    m = re.search(r"<%s[^>]*>(.*?)</%s>" % (name, name), text, re.S)
    return m.group(1) if m else ""


class Handler(http.server.BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"

    def log_message(self, fmt, *a):
        pass

    def _send(self, xml, code=200):
        body = xml.encode()
        self.send_response(code)
        self.send_header("Content-Type", "text/xml; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.send_header("Set-Cookie", "webvpn=MOCKSESSIONCOOKIE; path=/; secure")
        self.end_headers()
        self.wfile.write(body)

    def do_GET(self):
        sys.stderr.write("[mock] GET %s\n" % self.path)
        self._send(FORM)

    def do_POST(self):
        n = int(self.headers.get("Content-Length") or 0)
        raw = self.rfile.read(n) if n else b""
        text = raw.decode("utf-8", "replace")
        kind = tag(text, "config-auth") and re.search(r'type="(\w[\w-]*)"', text).group(1)
        sys.stderr.write("[mock] POST type=%s len=%d\n" % (kind, len(raw)))
        if kind == "init":
            self._send(FORM)
            return
        if kind == "config":
            sys.stderr.write("[mock]   -> CONFIG REPLY\n")
            self._send(CONFIG_REPLY)
            return

        user = tag(text, "username") or tag(text, "username")
        passwd = tag(text, "password")
        group = tag(text, "group-select") or tag(text, "auth_group")
        sys.stderr.write("[mock]   user=%r pass=%s group=%r\n"
                         % (user, "*" * len(passwd), group))
        if user and passwd == PASSWORD:
            sys.stderr.write("[mock]   -> AUTH OK\n")
            self._send(AUTH_OK)
        else:
            sys.stderr.write("[mock]   -> AUTH FAIL\n")
            self._send(AUTH_FAIL)


def main():
    ctx = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER)
    ctx.load_cert_chain("mock.crt", "mock.key")
    srv = http.server.ThreadingHTTPServer(("127.0.0.1", PORT), Handler)
    srv.socket = ctx.wrap_socket(srv.socket, server_side=True)
    sys.stderr.write("[mock] listening on https://127.0.0.1:%d\n" % PORT)
    srv.serve_forever()


if __name__ == "__main__":
    main()
